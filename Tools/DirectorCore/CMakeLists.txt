cmake_minimum_required(VERSION 3.16)
project(TheThugNativeDirector LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(NOT TARGET the_thug_party_save)
  add_subdirectory(../SaveCore ${CMAKE_CURRENT_BINARY_DIR}/SaveCore)
endif()
add_library(the_thug_director STATIC DirectorCore.cpp)
target_link_libraries(the_thug_director PUBLIC the_thug_party_save)
if(MSVC)
  target_compile_options(the_thug_director PRIVATE /utf-8)
endif()
add_executable(director_tests tests.cpp)
target_link_libraries(director_tests PRIVATE the_thug_director)
add_executable(director_parity_driver parity_driver.cpp)
target_link_libraries(director_parity_driver PRIVATE the_thug_director)
add_test(NAME director_data_fresh COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/build_data.py --check)
add_test(NAME director_core COMMAND director_tests)
add_test(NAME director_python_parity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/parity.py --driver $<TARGET_FILE:director_parity_driver>)
