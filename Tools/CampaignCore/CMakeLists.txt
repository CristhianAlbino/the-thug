cmake_minimum_required(VERSION 3.16)
project(TheThugCampaignRuntime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME campaign_data_fresh COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/build_data.py --check)
add_subdirectory(../WorldCore ${CMAKE_CURRENT_BINARY_DIR}/WorldCore)
add_library(the_thug_campaign STATIC CampaignCore.cpp CampaignSave.cpp)
target_link_libraries(the_thug_campaign PUBLIC the_thug_world)
add_executable(runtime_campaign_tests tests.cpp)
target_link_libraries(runtime_campaign_tests PRIVATE the_thug_campaign)
add_test(NAME runtime_campaign COMMAND runtime_campaign_tests)
