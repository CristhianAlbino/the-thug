cmake_minimum_required(VERSION 3.16)
project(TheThugCampaignRuntime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/utf-8)
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME campaign_data_fresh COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/build_data.py --check)
add_test(NAME companion_data_fresh COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/build_companion_data.py --check)
add_subdirectory(../WorldCore ${CMAKE_CURRENT_BINARY_DIR}/WorldCore)
add_subdirectory(../DirectorCore ${CMAKE_CURRENT_BINARY_DIR}/DirectorCore)
add_library(the_thug_campaign STATIC CampaignCore.cpp CampaignSave.cpp CampaignNarrative.cpp CampaignCompanions.cpp CampaignCompanionSave.cpp CheckpointFile.cpp CampaignHunter.cpp CampaignHunterSave.cpp CampaignCombat.cpp CampaignCombatSave.cpp CampaignSpecial.cpp CampaignSpecialSave.cpp)
target_link_libraries(the_thug_campaign PUBLIC the_thug_world the_thug_director)
add_executable(runtime_campaign_tests tests.cpp)
target_link_libraries(runtime_campaign_tests PRIVATE the_thug_campaign)
add_test(NAME runtime_campaign COMMAND runtime_campaign_tests)
add_executable(runtime_companion_tests companion_tests.cpp)
target_link_libraries(runtime_companion_tests PRIVATE the_thug_campaign)
add_executable(runtime_hunter_tests hunter_tests.cpp)
target_link_libraries(runtime_hunter_tests PRIVATE the_thug_campaign)
add_test(NAME runtime_hunter COMMAND runtime_hunter_tests)
foreach(suite IN ITEMS narrative_tests checkpoint_tests checkpoint_driver)
  add_executable(${suite} ${suite}.cpp)
  target_link_libraries(${suite} PRIVATE the_thug_campaign)
endforeach()
add_test(NAME runtime_companions COMMAND runtime_companion_tests)
add_test(NAME runtime_narrative COMMAND narrative_tests)
add_test(NAME runtime_checkpoint COMMAND checkpoint_tests)
add_test(NAME runtime_checkpoint_crashes COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/checkpoint_crash_tests.py $<TARGET_FILE:checkpoint_driver>)

add_executable(runtime_combat_tests combat_tests.cpp)
target_link_libraries(runtime_combat_tests PRIVATE the_thug_campaign)
add_test(NAME runtime_combat COMMAND runtime_combat_tests)

add_executable(runtime_special_tests special_tests.cpp)
target_link_libraries(runtime_special_tests PRIVATE the_thug_campaign)
add_test(NAME runtime_special COMMAND runtime_special_tests)
